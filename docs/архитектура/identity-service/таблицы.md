# app_user
Здесь хранится основаня информация об аккаунте

\*id: uuid PK
\*login: text, unique
\*password_hash: text
first_name: text
surname: text
patronymic:  text
policy_version: text - номер версии соглашения, которое принимал регистрируемый при регистрации
\*blocked: boolean - заблокирован пользовватель или нет
\*created_at: timestamp with time zone
\*name_requested: boolean - показывает, ввел ли родитель свои ФИО

# user_role
\*user_id: uuid PK FK → app_user(id)
\*role: text PK — ADMIN, TEACHER, PARENT, STUDENT

Составной первичный ключ: (user_id, role). У пользователя может быть несколько ролей.

# student_profile
\*user_id: uuid PK FK
\*password: text
school_grade: smallint

# student_reg_request
\*id: bigserial PK
\*parent_id - id родителя
\*first_name: text
\*surname: text
patronymic:  text
\*status: text
student_id: id зарегистрированного студента таблицы app_user после одобрения заявки 

# invitation_code
\*id: bigserial PK
\*code_hash: ttext
\*created_at: timestamp with time zone
\*expired_at: timestamp with time zone

# teacher_registration_code
\*id: bigserial PK
\*code_hash: ttext
\*created_at: timestamp with time zone
\*expired_at: timestamp with time zone

# parent_student
parent_id UUID NOT NULL REFERENCES app_user(id),
    student_id UUID NOT NULL REFERENCES app_user(id),
    PRIMARY KEY (parent_id, student_id)
